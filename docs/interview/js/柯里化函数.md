# 柯里化函数

## 概念

将接受 **多个参数的函数** 转换成一系列只接受 **单个参数的函数** 的技术

```js
// 普通函数
function add(a, b, c) {
  return a + b + c;
}
add(1, 2, 3); // 6

// 柯里化后的函数
function curriedAdd(a) {
  return function (b) {
    return function (c) {
      return a + b + c;
    };
  };
}
curriedAdd(1)(2)(3); // 6
```

- 核心思想：柯里化将 f(a, b, c) 转换为 f(a)(b)(c)

- 本质：利用闭包保存参数，延迟函数执行

- 优势：参数复用、函数组合、惰性执行

## 柯里化的应用场景

### 1. 参数复用

```js
// 未柯里化
function discount(price, discount) {
return price \* discount;
}
discount(100, 0.9); // 90
discount(200, 0.9); // 180

// 柯里化后
const curriedDiscount = curry(discount);
const tenPercentOff = curriedDiscount(0.9);
tenPercentOff(100); // 90
tenPercentOff(200); // 180
```

2. 延迟执行

```js
const curriedSum = curry((a, b, c) => a + b + c);
const add5 = curriedSum(5);
const add5and3 = add5(3);
// 可以在任何需要的时候调用
add5and3(2); // 10
```

3. 函数组合

```js
const map = curry((fn, arr) => arr.map(fn));
const filter = curry((fn, arr) => arr.filter(fn));

const double = x => x \* 2;
const isEven = x => x % 2 === 0;

const doubleEvens = arr => map(double)(filter(isEven)(arr));
doubleEvens([1, 2, 3, 4]); // [4, 8]
```

## 手动封装柯里化函数

```js
// 手动柯里化版本 (特定函数)
function curriedAdd(a) {
  return function (b) {
    return function (c) {
      return a + b + c;
    };
  };
}

console.log("手动柯里化:", curriedAdd(1)(2)(3)); // 6

// 通用柯里化函数 (面试必背)
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      // 参数够了，直接执行
      return fn.apply(this, args);
    } else {
      // 参数不够，返回新函数继续收集参数
      return function (...args2) {
        return curried.apply(this, args.concat(args2));
      };
    }
  };
}

// 测试通用柯里化
function add(a, b, c) {
  return a + b + c;
}

const curried = curry(add);

console.log(curried(1, 2, 3)); // 6
console.log(curried(1)(2, 3)); // 6
console.log(curried(1)(2)(3)); // 6
```
